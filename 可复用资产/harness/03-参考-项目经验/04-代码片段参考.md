## 1. 分布式锁（Redis + 唯一 token + Lua 释放）

```python
import asyncio, uuid
import redis.asyncio as aioredis

# 释放锁必须校验持有者：只有 token 匹配才删除（防误删他人锁）
_RELEASE_LUA = """
if redis.call("get", KEYS[1]) == ARGV[1] then
    return redis.call("del", KEYS[1])
else
    return 0
end
"""

async def acquire_lock(redis, key: str, ttl: int = 10) -> str:
    token = uuid.uuid4().hex
    ok = await redis.set(key, token, nx=True, px=ttl * 1000)
    return token if ok else ""

async def release_lock(redis, key: str, token: str):
    await redis.eval(_RELEASE_LUA, 1, key, token)
```

## 2. 乐观锁（余额变更 + 重试）

```python
async def _apply_balance_change(conn, user_id: int, delta: int, max_retries: int = 3) -> bool:
    """乐观锁更新余额；版本冲突重试。注意：余额与流水必须在同一事务。"""
    for _ in range(max_retries):
        row = await conn.fetchrow(
            "SELECT id, version, balance FROM user_wallets WHERE user_id=$1 FOR UPDATE", user_id
        )
        if not row:
            return False
        new_balance = row["balance"] + delta
        if new_balance < 0:
            return False  # 余额不足
        updated = await conn.execute(
            "UPDATE user_wallets SET balance=$1, version=version+1 "
            "WHERE user_id=$2 AND version=$3",
            new_balance, user_id, row["version"],
        )
        if updated.endswith("1"):  # 影响 1 行 = 成功
            return True
        # 否则版本冲突，重试
    return False
```

## 3. 幂等键（Redis 结果回放 + 唯一索引双保险）

```python
async def idempotent_create(redis, conn, idem_key: str, handler):
    """同一幂等键返回同一结果；Redis 存结果，DB 唯一索引兜底。"""
    cached = await redis.get(f"idem:{idem_key}")
    if cached:
        return cached.decode()
    # 先查库（唯一索引冲突即重放）
    result = await handler(conn)   # handler 内部用唯一索引
    await redis.set(f"idem:{idem_key}", result, ex=3600)
    return result
```

## 4. 滑动窗口限流（Redis ZSET，精确）

```python
async def allow(redis, key: str, limit: int, window: int) -> bool:
    """窗口内请求数上限；用 ZSET 时间戳实现，返回是否放行。"""
    now = redis.loop.time() * 1000 if hasattr(redis, "loop") else None
    import time
    now = time.time() * 1000
    pipe = redis.pipeline()
    pipe.zremrangebyscore(key, 0, now - window * 1000)
    pipe.zcard(key)
    pipe.zadd(key, {str(now): now})
    pipe.expire(key, window)
    _, count, _, _ = await pipe.execute()
    return count < limit
```

## 6. 缓存防击穿（互斥重建）

```python
async def get_or_rebuild(redis, key: str, rebuild, ttl: int = 300):
    """热点缓存失效时只允许一个请求重建，其余等待。"""
    val = await redis.get(key)
    if val:
        return val
    lock_key = f"lock:{key}"
    if await redis.set(lock_key, "1", nx=True, ex=10):
        try:
            val = await rebuild()
            await redis.set(key, val, ex=ttl)
            return val
        finally:
            await redis.delete(lock_key)
    # 未抢到锁：短暂等待后读缓存
    await asyncio.sleep(0.1)
    return await redis.get(key) or await rebuild()
```

## 7. 熔断（按端点隔离）

```python
class CircuitBreaker:
    """注册表模式：每个端点独立熔断，避免全局雪崩。"""
    def __init__(self, failure_threshold: int = 5, reset_after: int = 30):
        self.failures = 0
        self.state = "closed"   # closed / open / half-open
        self.opened_at = 0.0
        self.failure_threshold = failure_threshold
        self.reset_after = reset_after

    def allow(self) -> bool:
        if self.state == "open" and time.time() - self.opened_at > self.reset_after:
            self.state = "half-open"   # 试探放行一个请求
        return self.state != "open"

    def record(self, ok: bool):
        if ok:
            self.failures = 0
            self.state = "closed"
        else:
            self.failures += 1
            if self.failures >= self.failure_threshold:
                self.state = "open"
                self.opened_at = time.time()
```
