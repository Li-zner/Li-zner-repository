_AC_AUTOMATON = None

def _get_ac_automaton():
    global _AC_AUTOMATON
    if _AC_AUTOMATON is not None:
        return _AC_AUTOMATON
    if not AHOCORASICK_AVAILABLE:
        return None
    _AC_AUTOMATON = ahocorasick.Automaton()
    for idx, pattern in enumerate(_REASONING_LEAK_PATTERNS):
        _AC_AUTOMATON.add_word(pattern.lower(), (idx, pattern))
    _AC_AUTOMATON.make_automaton()
    return _AC_AUTOMATON

def _reasoning_leaked(text: str) -> bool:
    if not text:
        return False
    ac = _get_ac_automaton()
    if ac is not None:
        for _, _ in ac.iter(text.lower()):
            return True
        return False
    # 降级：遍历匹配
    low = text.lower()
    return any(p in low for p in _REASONING_LEAK_PATTERNS)

def _sanitize_reasoning(text: str, persona) -> str:
    """过滤思考内容；兼容字符串(如"me")和人格对象两种传参"""
    if not text:
        return ""
    # 1. 若传入字符串，兼容历史硬编码逻辑（"me" 不过滤）
    if isinstance(persona, str):
        if persona == "me":
            return text
    # 2. 若传入对象，按配置字段决定是否展示思考
    elif persona and not getattr(persona, 'show_reasoning', True):
        return ""
    # 3. 敏感词泄露检测
    if _reasoning_leaked(text):
        return ""
    return text.strip()

ALLOWED_EXTENSIONS = {
    '.txt', '.md', '.csv', '.json', '.xml', '.yaml', '.yml',
    '.log', '.html', '.css', '.sql', '.ini', '.cfg', '.conf', '.toml',
    '.rst', '.tex', '.rtf', '.pdf', '.jpg', '.jpeg', '.png', '.bmp', '.webp', '.docx'
}
MAX_FILE_SIZE = 20 * 1024 * 1024
MAX_CACHED_FILE_TEXT = 200 * 1024  # 修复 #4

# 固定上传目录（基于当前文件绝对路径）
UPLOAD_DIR = str(Path(__file__).resolve().parent.parent / "uploads")
Path(UPLOAD_DIR).mkdir(parents=True, exist_ok=True)

async def save_upload_file_chunked(file: UploadFile, save_path: str, chunk_size: int = 1024 * 1024) -> int:
    """分块流式保存，防止内存爆炸（修复 #20）"""
    total = 0
    loop = asyncio.get_running_loop()
    first = await file.read(chunk_size)
    if len(first) > MAX_FILE_SIZE:
        raise HTTPException(400, f"文件过大，最大支持 {MAX_FILE_SIZE//1024//1024}MB")
    def _write(data):
        with open(save_path, "ab") as f:
            f.write(data)
    await loop.run_in_executor(None, _write, first)
    total += len(first)
    while True:
        chunk = await file.read(chunk_size)
        if not chunk:
            break
        if total + len(chunk) > MAX_FILE_SIZE:
            try:
                await loop.run_in_executor(None, lambda: os.remove(save_path))
            except FileNotFoundError:
                pass
            raise HTTPException(400, f"文件过大，最大支持 {MAX_FILE_SIZE//1024//1024}MB")
        await loop.run_in_executor(None, _write, chunk)
        total += len(chunk)
    return total

async def extract_text_content_async(filepath: str, filename: str) -> str:
    ext = os.path.splitext(filename)[1].lower()
    if ext not in ALLOWED_EXTENSIONS:
        raise ValueError(f"不支持的文件类型: {ext}")
    loop = asyncio.get_running_loop()
    def _read():
        try:
            with open(filepath, 'r', encoding='utf-8') as f:
                return f.read()
        except UnicodeDecodeError:
            with open(filepath, 'r', encoding='gbk') as f:
                return f.read()
    content = await loop.run_in_executor(None, _read)
    if len(content) > MAX_CACHED_FILE_TEXT:
        content = content[:MAX_CACHED_FILE_TEXT] + "\n...(内容过长，已截断)"
    return content

@router.post("/upload", response_model=FileUploadResponse)
async def upload_file(
    file: UploadFile = File(...),
    current_user: dict = Depends(get_current_user)
):
    safe_filename = os.path.basename(file.filename or "")
    if not safe_filename:
        raise HTTPException(400, "文件名无效")
    ext = os.path.splitext(safe_filename)[1].lower()
    if ext not in ALLOWED_EXTENSIONS:
        raise HTTPException(400, f"不支持的文件类型 '{ext}'")
    file_id = str(uuid.uuid4())
    save_path = os.path.join(UPLOAD_DIR, f"{file_id}{ext}")
    try:
        file_size = await save_upload_file_chunked(file, save_path)
    except HTTPException as e:
        loop = asyncio.get_running_loop()
        try:
            await loop.run_in_executor(None, lambda: os.remove(save_path))
        except FileNotFoundError:
            pass
        raise e

    BINARY_EXTS = {'.pdf', '.jpg', '.jpeg', '.png', '.bmp', '.webp', '.docx'}
    if ext in BINARY_EXTS:
        from ..agents.document_parser import parse_document
        result = await parse_document(save_path, safe_filename)
        if result["success"]:
            text_content = result["text"]
            parse_note = f"（{result['format']}，已提取）"
        else:
            text_content = result.get("error", "解析失败")
            parse_note = f"（{result['format']}，提取失败）"
    else:
        try:
            text_content = await extract_text_content_async(save_path, safe_filename)
            parse_note = "（文本文件）"
        except ValueError as e:
            loop = asyncio.get_running_loop()
            try:
                await loop.run_in_executor(None, lambda: os.remove(save_path))
            except FileNotFoundError:
                pass
            raise HTTPException(400, detail=str(e))

    r = await get_redis()
    file_meta = {
        "file_id": file_id,
        "filename": safe_filename,
        "ext": ext,
        "size": file_size,
        "uploaded_by": current_user["username"],
        "uploaded_at": datetime.now().isoformat(),
        "text_length": len(text_content),
        "parse_note": parse_note,
    }
    await r.setex(f"file:{file_id}:content", 3600, text_content)
    await r.setex(f"file:{file_id}:meta", 3600, json.dumps(file_meta))
    logger.info(f"📎 上传成功: {safe_filename}, size={file_size}, text_len={len(text_content)}")
    return FileUploadResponse(
        **file_meta,
        text_preview=text_content[:2000] + ("..." if len(text_content) > 2000 else ""),
        text_content=text_content[:50000]
    )

async def fallback_flash(req: ChatRequest, username: str):
    from ..core.metrics import gateway_errors_total
    gateway_errors_total.labels(status='fallback_flash').inc()
    logger.info(f"⚡ Flash降级: {username}")
    key = _get_deepseek_key()
    if not key:
        return
    try:
        async with llm_semaphore:
            async with httpx.AsyncClient(timeout=HTTP_TIMEOUT_MEDIUM) as client:
                async with client.stream(
                    "POST",
                    f"{DEEPSEEK_API_BASE}/chat/completions",
                    headers={"Authorization": f"Bearer {key}", "Content-Type": "application/json"},
                    json={
                        "model": DEEPSEEK_FLASH_MODEL,
                        "messages": [
                            {"role": "system", "content": "你是AI助手，直接回答。" + _lang_instruction(req)},
                            {"role": "user", "content": req.query}
                        ],
                        "stream": True,
                        "temperature": 0.3,
                        "max_tokens": 2048
                    }
                ) as resp:
                    resp.raise_for_status()
                    async for line in resp.aiter_lines():
                        if line.startswith("data:"):
                            data_str = line[5:].strip()
                            if data_str == "[DONE]":
                                yield "data: [DONE]\n\n"
                                return
                            try:
                                data = jloads(data_str)
                                chunk = data.get("choices", [{}])[0].get("delta", {}).get("content", "")
                                if chunk:
                                    from ..core.safety_filter import get_filter
                                    sf = get_filter()
                                    result = sf.check_stream(chunk)
                                    if not result['safe']:
                                        logger.warning(f"🛡️ DFA拦截: {result['triggered_word']}")
                                        yield sse("answer_chunk", sf.safe_message)
                                        yield sse("answer_complete", sf.safe_message) + "data: [DONE]\n\n"
                                        return
                                    yield sse("answer_chunk", chunk)
                            except:
                                pass
                    yield "data: [DONE]\n\n"
    except Exception as e:
        logger.warning(f"Flash降级失败: {e}")

# ========== 流式节流辅助（修复 #16） ==========
async def _stream_with_throttle(generator, persona):
    """对 LLM 流式输出进行节流，同时过滤思考内容"""
    full = ""
    buf = ""
    first_time = None
    last_time = None
    async for ev in generator:
        if ev["type"] == "reasoning":
            chunk = _sanitize_reasoning(ev["text"], persona)
            if chunk:
                yield sse("reasoning_chunk", chunk)
        elif ev["type"] == "content":
            chunk = ev["text"]
            full += chunk
            buf += chunk
            now = time.time()
            if first_time is None:
                first_time = last_time = now
                yield sse("answer_chunk", chunk)
                buf = ""
            elif len(buf) >= 40 or (now - last_time >= 0.1 and buf):
                yield sse("answer_chunk", buf)
                buf = ""
                last_time = now
        elif ev["type"] == "usage":
            yield ev
    if buf:
        yield sse("answer_chunk", buf)
    return full

# ========== 限流检查（辅助） ==========
async def _check_rate_limits(username, user_role, today):
    from ..middleware.rate_limit import _ROLE_LIMITS
    if not await check_qps(username, user_role):
        gateway_requests_total.labels(method='POST', endpoint='/v2/chat/stream', status='429').inc()
        raise HTTPException(429, "请求过于频繁")
    if not await check_concurrent(username, user_role):
        await release_concurrent(username)
        raise HTTPException(429, "并发过多")
    usage = await get_daily_usage(username, today)
    limits = _ROLE_LIMITS.get(user_role, {"daily_req": DAILY_REQUEST_LIMIT, "daily_token": DAILY_TOKEN_LIMIT})
    if usage["request_count"] >= limits["daily_req"]:
        await release_concurrent(username)
        raise HTTPException(429, "今日请求已达上限")
    if usage["token_sum"] >= limits["daily_token"]:
        await release_concurrent(username)
        raise HTTPException(429, "今日Token已达上限")
    return usage

# ========== 缓存拦截（辅助） ==========
async def _try_cache_hit(req, lang_instr):
    if req.file_ids:
        return False, None
    try:
        cached = await SemanticCache.get(req.query)
        if cached and await SemanticCache.is_empty(cached):
            msg = "服务暂时繁忙，请稍后重试。" if getattr(req, 'lang', 'zh')!='en' else "Service busy, try later."
            return True, msg
        if cached:
            from ..core.safety_filter import get_filter
            sf = get_filter()
            if sf.contains_sensitive(cached):
                cached = sf.safe_message
            return True, cached
    except Exception:
        pass
    return False, None

# ========== system prompt 构建（修复 #3） ==========
def _build_system_prompt(persona, req, lang_instr, shared_context=""):
    today_str = datetime.now().strftime("%Y年%m月%d日 %A")
    if persona:
        try:
            content = persona.system_prompt.format(today=today_str, name=persona.name)
        except KeyError:
            content = persona.system_prompt.replace("{today}", today_str).replace("{name}", persona.name)
    else:
        content = f"你是AI助手。今天是{today_str}。"
    if shared_context:
        content += f"\n{shared_context}"
    content += lang_instr
    return content

                    )
                except asyncio.TimeoutError:
                    logger.warning("工具并行调用超时，触发降级")
                    for task in tasks:
                        if not task.done():
                            task.cancel()
                    yield sse("reasoning_chunk", '知识库查询超时，正在基于已有知识继续回答...')
                    tool_results = []
                    for tc in tool_calls:
                        tool_results.append({"error": "工具查询超时", "fallback": True})

                for idx, result in enumerate(tool_results):
                    if isinstance(result, Exception):
                        result = {"error": str(result)}
                    yield f"data: {json.dumps({'type': 'tool_result', 'index': idx, 'result': result})}\n\n"

                # 法律映射表命中检查（工具返回中可能含 mapping_hit）
                _law_mapping_hit = None
                for _res in tool_results:
                    if isinstance(_res, dict) and _res.get("mapping_hit"):
                        _law_mapping_hit = _res.get("message", "")
                        break
                if _law_mapping_hit:
                    logger.info(f"⚖️ 法律映射表命中，跳过LLM生成")
                    yield sse("answer_complete", _law_mapping_hit) + "data: [DONE]\n\n"
                    await mm.save_messages(
                        {"role": "user", "content": req.query},
                        {"role": "assistant", "content": _law_mapping_hit}
                    )
                    await SemanticCache.set(req.query, _law_mapping_hit)
                    await update_daily_usage(username, today, inc_request=1, inc_token=0)
                    asyncio.create_task(inc_used_questions(username))
                    saved_normally = True
                    return

                # 多 Agent 圆桌讨论
                orchestrator = None
                try:
                    from ..agents.orchestrator import AgentOrchestrator
                    orch = AgentOrchestrator(req.query, req.user_location or "")
                    agent_names = get_agent_names_for_orchestrator(matched_agents) if 'get_agent_names_for_orchestrator' in locals() else ["query_weather", "query_hotel", "query_route", "query_food"]
                    if not agent_names:
                        agent_names = ["query_weather", "query_hotel", "query_route", "query_food"]
                    for idx, tc in enumerate(tool_calls):
                        func_name = tc["function"]["name"]
                        result = tool_results[idx] if idx < len(tool_results) else {"error": "missing result"}
                        if isinstance(result, Exception):
                            result = {"error": str(result)}
                        if func_name in agent_names:
                            if isinstance(result, dict) and "error" in result:
                                orch.add_tool_result(func_name, {}, error=str(result["error"]))
                            else:
                                orch.add_tool_result(func_name, result)
                    if orch.get_involved_agents() and len(agent_names) > 1:
                        yield sse("thought", '专家们正在讨论分析...')
                        discussion_summary = await orch.run(enable_phase2=False)
                        orchestrator = orch
                        logger.info(f"🧠 多Agent讨论完成（{len(agent_names)}个Agent参与）")
                    else:
                        discussion_summary = ""
                except Exception as orch_err:
                    logger.warning(f"多Agent讨论异常: {orch_err}")
                    discussion_summary = ""

                # 将工具结果拼回消息
                tool_messages = []
                for idx, result in enumerate(tool_results):
                    if isinstance(result, Exception):
                        result = {"error": str(result)}
                    if isinstance(result, dict) and "error" in result:
                        error_msg = result["error"]
                        fallback_text = "获取数据失败了，可能服务暂时不可用。"
                        if "未找到" in error_msg:
                            fallback_text = "暂时没找到这个城市的数据，建议换个关键词试试～"
                        elif "超时" in error_msg:
                            fallback_text = "查询有点慢，可能网络问题，请稍后再试～"
                        elif "Key" in error_msg or "授权" in error_msg:
                            fallback_text = "服务配置正在更新，暂时无法使用，我试试其他方式帮你。"
                        result = {
                            "error": error_msg,
                            "fallback_message": fallback_text,
                            "success": False
                        }
                    tool_messages.append({
                        "role": "tool",
                        "tool_call_id": tool_calls[idx]["id"],
                        "content": json.dumps(result, ensure_ascii=False)
                    })

                messages.append(message)
                messages.extend(tool_messages)

                if discussion_summary:
                    for line in discussion_summary.split('\n'):
                        if line.strip():
                            yield sse("reasoning_chunk", line + "\n")
                    yield f"data: {json.dumps({'type': 'reasoning_done'})}\n\n"

                    messages.append({
                        "role": "system",
                        "content": (
                            "以下是基于工具查询结果的专业分析，已作为内部参考提供给你。\n"
                            "请**直接**使用这些信息来回答用户，形成一份完整、连贯、自然的回答。\n"
                            "🚨 **禁止在回答中提及**「专家」「天气专家说」「据讨论」「根据分析」等词汇。\n"
                            "不要引用任何专家意见，不要用「某某专家认为」的句式。\n"
                            "用你自己的口吻，把这些信息整合成一段流畅的建议。\n\n"
                            f"{discussion_summary}"
                        )
                    })

            # 循环结束未返回（max_steps 耗尽）
            if not saved_normally:
                last_content = ""
                for m in reversed(messages):
                    if m.get("role") == "assistant" and m.get("content"):
                        last_content = m["content"]
                        break
                if not last_content:
                    last_content = full_content or "抱歉，我暂时无法完成完整的回答。"
                logger.warning(f"V2 循环达到最大步数，返回已有内容: {len(last_content)} chars")
                yield sse("answer_complete", last_content) + "data: [DONE]\n\n"
                partial_answer = last_content
                await mm.save_messages(
                    {"role": "user", "content": req.query},
                    {"role": "assistant", "content": last_content}
                )
                await SemanticCache.set(req.query, last_content)
                await update_daily_usage(username, today, inc_request=1, inc_token=0)
                asyncio.create_task(inc_used_questions(username))
                saved_normally = True

        except Exception as e:
            logger.error(f"V2 未知异常，降级备胎: {e}", exc_info=True)
            try:
                async for chunk in fallback_chain(req, username):
                    yield chunk
            except Exception as fb_err:
                logger.error(f"降级也失败: {fb_err}")
                yield sse("answer_complete", '服务暂时不可用，请稍后再试。') + "data: [DONE]\n\n"
        finally:
            if _rebuild_lock:
                try:
                    await SemanticCache.release_rebuild_lock(req.query, _rebuild_lock)
                except Exception as _rel_err:
                    logger.warning(f"防击穿锁释放失败（TTL 自愈兜底）: {_rel_err}")
            await release_concurrent(username)
            if not saved_normally and mm and partial_answer and partial_answer != "抱歉，我暂时无法回答。":
                try:
                    await mm.save_messages(
                        {"role": "user", "content": req.query},
                        {"role": "assistant", "content": partial_answer + "\n\n（内容不完整，连接已断开）"}
                    )
                    logger.info(f"💾 断线保存: conv_id={conv_id}, 内容长度={len(partial_answer)}")
                    await SemanticCache.set(req.query, partial_answer)
                except Exception as save_err:
                    logger.warning(f"断线保存失败: {save_err}")

    return StreamingResponse(generate(), media_type="text/event-stream")

@router.post("/chat/tasks", status_code=201)
async def create_chat_task(
    req: CreateTaskRequest,
    current_user: dict = Depends(get_current_user)
):
    username = current_user["username"]

    if current_user.get("role") != "admin" and is_quota_exhausted(current_user):
        raise HTTPException(status_code=402, detail="免费额度已用完，请绑定手机号后继续使用")

    # 修复 #5：幂等键包含 username
    existing = await try_idempotent(username, req.session_id, req.message)
    if existing:
        return {"task_id": existing, "idempotent": True}

    # 修复 #6：存储时保留 persona_id 和 file_ids
    task_id = await create_task(
        req.session_id,
        req.message,
        req.user_location,
        persona_id=req.persona_id,
        file_ids=req.file_ids
    )
    save_idempotent(username, req.session_id, req.message, task_id)

    # 修复 #18：通过 llm_semaphore 控制并发（使用 asyncio.create_task 但受 semaphore 限制）
    # 注意：这里不阻塞，因为 run_agent_task 内部会使用 llm_semaphore
    asyncio.create_task(run_agent_task(
        task_id=task_id,
        username=username,
        session_id=req.session_id,
        user_query=req.message,
        user_location=req.user_location,
        persona_id=req.persona_id,
        file_ids=req.file_ids,
    ))

    return {"task_id": task_id, "idempotent": False}


@router.get("/chat/tasks/{task_id}/result", response_model=TaskResultResponse)
async def get_task_result(
    task_id: str,
    wait: int = 0,
    current_user: dict = Depends(get_current_user)
):
    """查询/等待任务结果"""
    task = await get_task(task_id)
    if not task:
        from fastapi.responses import JSONResponse
        return JSONResponse({"error": "task not found"}, status_code=404)

    status = task["status"]

    if status in ("completed", "cancelled", "error"):
        result = await read_accumulated_result(task_id) or task.get("result", "")
        return TaskResultResponse(
            status=status,
            content=result,
            session_id=task.get("session_id", ""),
        )

    # 修复 #7：使用 wait 参数控制最大等待时长
    if wait and status in ("pending", "generating"):
        max_wait = min(wait, 60) if wait > 0 else 0
        deadline = time.time() + max_wait
        while time.time() < deadline:
            await asyncio.sleep(0.5)
            task = await get_task(task_id)
            if not task:
                return TaskResultResponse(status="error", content="task not found")
            st = task["status"]
            if st in ("completed", "cancelled", "error"):
                result = await read_accumulated_result(task_id) or task.get("result", "")
                return TaskResultResponse(status=st, content=result, session_id=task.get("session_id", ""))
            partial = await read_accumulated_result(task_id)
            if partial:
                return TaskResultResponse(status="generating", content=partial, session_id=task.get("session_id", ""))

    partial = await read_accumulated_result(task_id) or ""
    return TaskResultResponse(status=status, content=partial, session_id=task.get("session_id", ""))


@router.post("/chat/tasks/{task_id}/cancel")
async def cancel_task(
    task_id: str,
    current_user: dict = Depends(get_current_user)
):
    task = await get_task(task_id)
    if not task:
        from fastapi.responses import JSONResponse
        return JSONResponse({"error": "task not found"}, status_code=404)

    if task["status"] in ("completed", "cancelled", "error"):
        return {"status": task["status"], "message": "任务已终结，无需取消"}

    set_cancelled(task_id)
    partial = await read_accumulated_result(task_id)
    await update_status(task_id, "cancelled", partial)

    return {"status": "cancelled", "message": "已取消"}


@router.post("/chat/tasks/{task_id}/resume")
async def resume_task(
    task_id: str,
    current_user: dict = Depends(get_current_user)
):
    """重新生成（创建新任务，丢弃旧草稿）"""
    task = await get_task(task_id)
    if not task:
        from fastapi.responses import JSONResponse
        return JSONResponse({"error": "task not found"}, status_code=404)

    if task["status"] != "cancelled":
        return {"error": "只有已取消的任务才能恢复", "status": task["status"]}

    username = current_user["username"]
    # 修复 #6：创建新任务时透传 persona_id 和 file_ids
    new_task_id = await create_task(
        task["session_id"],
        task["user_message"],
        task.get("user_location", ""),
        persona_id=task.get("persona_id"),
        file_ids=task.get("file_ids", [])
    )

    asyncio.create_task(run_agent_task(
        task_id=new_task_id,
        username=username,
        session_id=task["session_id"],
        user_query=task["user_message"],
        user_location=task.get("user_location", ""),
        persona_id=task.get("persona_id"),
        file_ids=task.get("file_ids", []),
    ))

    return {"task_id": new_task_id, "previous_task_id": task_id}
